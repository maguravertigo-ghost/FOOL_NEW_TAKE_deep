import { ASSETS, svgInstance } from '../assets.js';
import {
  CARD_W, CARD_H,
  CORNER_RANK, CORNER_SUIT,
  CORNER_RANK_SCALE, CORNER_SUIT_SCALE,
  CENTER_SUIT_SCALE, rankColor
} from './constants.js';

export function buildCardFace(data) {
  const face = document.createElement('div');
  face.className = 'card__layer card__face';
  face.style.width  = CARD_W + 'px';
  face.style.height = CARD_H + 'px';

  // ---- Тело карты (рамка) ----
  const body = makeSvgOrImg(ASSETS.body);
  if (body) {
    body.classList.add('card__body');
    setBoxSize(body, '100%', '100%');
    face.appendChild(body);
  }

  const rankSrc = ASSETS.ranks[data.rank];
  const suitSrc = ASSETS.suits[data.suit];
  const color   = rankColor(data.suit);

  // ---- Углы LT ----
  appendAt(face, makeSuit(suitSrc),
    CORNER_SUIT.x, CORNER_SUIT.y, false, CORNER_SUIT_SCALE);
  appendAt(face, makeRank(rankSrc, color),
    CORNER_RANK.x, CORNER_RANK.y, false, CORNER_RANK_SCALE);

  // ---- Углы RB (перевёрнуты) ----
  appendAt(face, makeSuit(suitSrc),
    CARD_W - CORNER_SUIT.x, CARD_H - CORNER_SUIT.y, true, CORNER_SUIT_SCALE);
  appendAt(face, makeRank(rankSrc, color),
    CARD_W - CORNER_RANK.x, CARD_H - CORNER_RANK.y, true, CORNER_RANK_SCALE);

  // ---- Центр: увеличенная масть ----
  const center = makeSuit(suitSrc);
  if (center) {
    center.style.position = 'absolute';
    center.style.left = (CARD_W / 2) + 'px';
    center.style.top  = (CARD_H / 2) + 'px';
    center.style.transformOrigin = 'center center';
    center.style.transform =
      `translate(-50%, -50%) scale(${CENTER_SUIT_SCALE})`;
    center.style.display = 'block';
    face.appendChild(center);
  }

  return face;
}

// ---------- helpers ----------

function makeSuit(url) {
  return makeSvgOrImg(url);
}

function makeRank(url, color) {
  const svg = svgInstance(url);
  if (svg) {
    tintSvg(svg, color);
    return svg;
  }
  // Фолбэк — img. Тонировать не получится, но увидим хотя бы форму.
  const img = document.createElement('img');
  img.src = url;
  img.alt = '';
  img.style.display = 'block';
  return img;
}

/**
 * Возвращает <svg> из кэша или <img> как фолбэк.
 * Возвращает null, если узел не создать (маловероятно).
 */
function makeSvgOrImg(url) {
  const svg = svgInstance(url);
  if (svg) return svg;

  const img = document.createElement('img');
  img.src = url;
  img.alt = '';
  img.style.display = 'block';
  return img;
}

/** Абсолютно позиционирует узел по (cx, cy) через translate(-50%,-50%). */
function appendAt(parent, node, cx, cy, flip, scale = 1) {
  if (!node) return;

  node.style.position = 'absolute';
  node.style.left = cx + 'px';
  node.style.top  = cy + 'px';
  node.style.transformOrigin = 'center center';

  const parts = ['translate(-50%, -50%)'];
  if (flip) parts.push('rotate(180deg)');
  if (scale !== 1) parts.push(`scale(${scale})`);
  node.style.transform = parts.join(' ');

  node.style.display = 'block';

  if (node.tagName.toLowerCase() === 'svg') {
    applyIntrinsicSize(node);
  }
  parent.appendChild(node);
}

/** Форсирует размер контейнера (для body: 100%×100%). */
function setBoxSize(node, w, h) {
  node.setAttribute('width',  typeof w === 'number' ? w : w);
  node.setAttribute('height', typeof h === 'number' ? h : h);
  node.setAttribute('preserveAspectRatio', 'none');
  node.style.width  = typeof w === 'number' ? w + 'px' : w;
  node.style.height = typeof h === 'number' ? h + 'px' : h;
  node.style.display = 'block';
}

/**
 * Если у <svg> нет явных width/height, но есть viewBox — ставим их
 * в пикселях из viewBox, иначе браузер раздувает SVG до 300×150.
 */
function applyIntrinsicSize(svg) {
  let w = parseFloat(svg.getAttribute('width'));
  let h = parseFloat(svg.getAttribute('height'));
  const vb = svg.getAttribute('viewBox');

  if ((!w || !h) && vb) {
    const parts = vb.split(/[\s,]+/).map(Number);
    if (parts.length === 4) {
      if (!w) w = parts[2];
      if (!h) h = parts[3];
    }
  }
  if (w && h) {
    svg.setAttribute('width',  w);
    svg.setAttribute('height', h);
    svg.style.width  = w + 'px';
    svg.style.height = h + 'px';
  }
}

/**
 * Перекрашивает заливки/обводки SVG. Прозрачные и градиентные не трогает.
 */
function tintSvg(svg, color) {
  const SHAPES =
    'path, rect, circle, ellipse, polygon, polyline, line, use, text, tspan';

  svg.querySelectorAll(SHAPES).forEach(el => {
    const aFill = el.getAttribute('fill');
    const sFill = (el.style.fill || '').toLowerCase();
    if (aFill === 'none' || sFill === 'none') return;
    if (aFill && aFill.startsWith('url(')) return;
    el.style.fill = color;

    const aStroke = el.getAttribute('stroke');
    const sStroke = (el.style.stroke || '').toLowerCase();
    if (aStroke && aStroke !== 'none' && sStroke !== 'none' &&
        !aStroke.startsWith('url(')) {
      el.style.stroke = color;
    }
  });

  svg.style.color = color;
  svg.setAttribute('color', color);
}
