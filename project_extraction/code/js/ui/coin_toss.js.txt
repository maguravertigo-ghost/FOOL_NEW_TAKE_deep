import { ASSETS } from '../assets.js';
import { SceneLayout } from '../game/scene_layout.js';
import { Progress } from '../game/progress.js';
import { uiScale } from './ui_scale.js';

/* Размер монеты в ref-px (231×246 × масштаб 0.5). */
const COIN_W_REF = 231 * 0.5;
const COIN_H_REF = 246 * 0.5;

export class CoinToss {
  constructor(app) {
    this.app = app;
    this.scene = app.scene;
    this.hud = app.menu.hud;

    this._layerEl = null;
    this._stripEl = null;
    this._labelEl = null;
    this._coinEl  = null;
    this._tokenEl = null;

    this._cancelled = false;
    this._speedMult = 1;
    this._frameIndex = 0;
    this._coinW = 0;
    this._coinH = 0;
    this._coinX = 0;
    this._coinY = 0;
	
	this._cornerSide = null;   // 'cpu' | 'player' | null
    this._flying = false;
    this._resizeUnsub = this.scene.onResize(() => this._onSceneResize());
  }

  /* ============================================================
   *  Публичный API
   * ============================================================ */

  /**
   * Полный цикл жеребьёвки: полоса + монета + вращение + результат.
   * Возвращает 'player' | 'cpu'.
   */
  async run() {
    this._createLayer();
    this._createStrip('ЖЕРЕБЬЁВКА');
    this._createCoin();

    const W = this.scene.width, H = this.scene.height;
    this._coinX = W / 2;
    this._coinY = H / 2;
    this._placeCoin();

    this._layerEl.style.opacity = '0';
    await this._fadeIn(600);

    // результат предопределён заранее — монета «доедет» до нужного кадра
    const result = Math.random() < 0.5 ? 'cpu' : 'player';
    const targetFrame = (result === 'cpu') ? 0 : 4;

    await this._spin(targetFrame);

    const labelText = (result === 'cpu')
      ? 'ПЕРВЫЙ ХОД ДЕЛАЕТ: ДЖОКЕР'
      : 'ПЕРВЫЙ ХОД ДЕЛАЕТ: ИГРОК';
    this._setStripLabel(labelText);

    await this._sleep(1500);
    if (this._cancelled) return result;

    await this._flyToCorner(result);
    await this._fadeStripOut(400);

	this._cornerSide = result;

    this._layerEl.style.pointerEvents = 'none';
    return result;
  }

  /**
   * Матч с уже определённым жетоном.
   * Показывает жетон в центре, летит в угол владельца.
   */
  async announceToken(token) {
    if (!token) return;
    this._createLayer();
    this._createToken();

    const W = this.scene.width, H = this.scene.height;
    this._coinX = W / 2;
    this._coinY = H / 2;
    this._placeToken();

    this._layerEl.style.opacity = '0';
    await this._fadeIn(600);

    await this._sleep(500);
    if (this._cancelled) return;

    await this._flyTokenToCorner(token.owner);
	
	this._cornerSide = token.owner;
	
    this._layerEl.style.pointerEvents = 'none';
  }

  destroy() {
    this._cancelled = true;
    if (this._resizeUnsub) {
      this._resizeUnsub();
      this._resizeUnsub = null;
    }
    if (this._layerEl && this._layerEl.parentNode) {
      this._layerEl.remove();
    }
    this._layerEl = null;
	this._cornerSide = null;
  }

  /* ============================================================
   *  DOM
   * ============================================================ */

  _createLayer() {
    const el = document.createElement('div');
    el.className = 'coin-toss-layer';
    el.addEventListener('click', () => {
      this._speedMult = Math.min(this._speedMult + 0.8, 4);
    });
    this.app.sceneEl.appendChild(el);
    this._layerEl = el;
  }

  _createStrip(text) {
    const strip = document.createElement('div');
    strip.className = 'coin-toss-strip';

    const label = document.createElement('div');
    label.className = 'coin-toss-strip__label';
    label.textContent = text;

    strip.appendChild(label);
    this._layerEl.appendChild(strip);
    this._stripEl = strip;
    this._labelEl = label;
  }

  _setStripLabel(text) {
    if (this._labelEl) this._labelEl.textContent = text;
  }

  _createCoin() {
    const img = document.createElement('img');
    img.className = 'coin-toss-coin';
    img.src = ASSETS.coinFrames[0];
    img.alt = '';
    img.draggable = false;

    const W = this.hud.s(COIN_W_REF);
    const H = this.hud.s(COIN_H_REF);
    this._coinW = W;
    this._coinH = H;
    img.style.width = W + 'px';
    img.style.height = H + 'px';

    this._layerEl.appendChild(img);
    this._coinEl = img;
  }

  _createToken() {
    const img = document.createElement('img');
    img.className = 'coin-toss-token';
    img.src = ASSETS.foolToken;
    img.alt = '';
    img.draggable = false;

    const W = this.hud.s(COIN_W_REF);
    const H = this.hud.s(COIN_H_REF);
    this._coinW = W;
    this._coinH = H;
    img.style.width = W + 'px';
    img.style.height = H + 'px';

    this._layerEl.appendChild(img);
    this._tokenEl = img;
  }

  _placeCoin() {
    if (!this._coinEl) return;
    this._coinEl.style.left = (this._coinX - this._coinW / 2) + 'px';
    this._coinEl.style.top  = (this._coinY - this._coinH / 2) + 'px';
  }

  _placeToken() {
    if (!this._tokenEl) return;
    this._tokenEl.style.left = (this._coinX - this._coinW / 2) + 'px';
    this._tokenEl.style.top  = (this._coinY - this._coinH / 2) + 'px';
  }

  /* ============================================================
   *  Анимации
   * ============================================================ */

  _fadeIn(duration) {
    return new Promise(resolve => {
      const el = this._layerEl;
      if (!el) return resolve();
      const t0 = performance.now();
      const tick = () => {
        if (this._cancelled) return resolve();
        const p = Math.min(1, (performance.now() - t0) / duration);
        el.style.opacity = String(p * p);   // ease-in
        if (p >= 1) return resolve();
        requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
    });
  }

  async _spin(targetFrame) {
    // фаза 1: раскрутка (медленно → быстро)
    const spinUpFrames = [150, 120, 100, 85, 70, 60, 50, 45];
    for (const d of spinUpFrames) {
      if (this._cancelled) return;
      this._nextFrame();
      await this._sleep(d);
    }

    // фаза 2: максимальная скорость
    const fastFrameDur = 40;
    const fastT0 = performance.now();
    const fastTotal = 1200;
    while ((performance.now() - fastT0) * this._speedMult < fastTotal) {
      if (this._cancelled) return;
      this._nextFrame();
      await this._sleep(fastFrameDur);
    }

    // фаза 3: замедление с посадкой на нужный кадр
    await this._spinDownTo(targetFrame);
  }

  async _spinDownTo(targetFrame) {
    const cur = this._frameIndex;
    let steps = (targetFrame - cur + 8) % 8;
    if (steps === 0) steps = 8;   // если уже на нужном — полный оборот

    const durs = [];
    for (let i = 0; i < steps; i++) {
      const p = steps > 1 ? i / (steps - 1) : 0;
      const k = p * p * (3 - 2 * p);      // ease-in-out
      durs.push(60 + (500 - 60) * k);
    }

    for (let i = 0; i < steps; i++) {
      if (this._cancelled) return;
      this._nextFrame();
      await this._sleep(durs[i]);
    }

    this._setFrame(targetFrame);
  }

  _nextFrame() {
    this._frameIndex = (this._frameIndex + 1) % ASSETS.coinFrames.length;
    if (this._coinEl) this._coinEl.src = ASSETS.coinFrames[this._frameIndex];
  }

  _setFrame(idx) {
    this._frameIndex = idx;
    if (this._coinEl) this._coinEl.src = ASSETS.coinFrames[idx];
  }

  async _flyToCorner(side) {
    return this._flyToCornerGeneric(side, this._coinEl);
  }

  async _flyTokenToCorner(side) {
    return this._flyToCornerGeneric(side, this._tokenEl);
  }

  async _flyToCornerGeneric(side, el) {
    if (!el) return;
 
    this._flying = true;
 
    // пересчитываем размер в текущем масштабе (F11 мог случиться до старта)
    this._measureCoinSize();
    el.style.width  = this._coinW + 'px';
    el.style.height = this._coinH + 'px';

    const W = this.scene.width, H = this.scene.height;
    const C = SceneLayout.config;
    const frameT = C.frameT * H;
    const frameR = W - C.frameR * W;
    const frameB = H - C.frameB * H;

    const cw = this._coinW, ch = this._coinH;

    // Финальный масштаб: монета уменьшается до 0.8 от исходного
    const kEnd  = 0.8;
    const cwEnd = cw * kEnd;
    const chEnd = ch * kEnd;

    let endX, endY;
    if (side === 'cpu') {
      endX = frameR - cwEnd / 2;
      endY = frameT + chEnd / 2;
    } else {
      endX = frameR - cwEnd / 2;
      endY = frameB - chEnd / 2;
    }

    const startX = this._coinX, startY = this._coinY;
    const dur = 900;
    const t0 = performance.now();

    return new Promise(resolve => {
      const tick = () => {
        if (this._cancelled) return resolve();
        const now = performance.now();
        const p = Math.min(1, (now - t0) * this._speedMult / dur);

        // X: ease-out
        const easeOut = 1 - (1 - p) * (1 - p);
        const x = startX + (endX - startX) * easeOut;

        // Y: с задержкой, ease-in
        const pY = Math.max(0, (p - 0.2) / 0.8);
        const easeInY = pY * pY;
        const y = startY + (endY - startY) * easeInY;

        // Масштаб: 1.0 → 0.8 синхронно с easing X
        const k = 1 + (kEnd - 1) * easeOut;

        el.style.left = (x - cw / 2) + 'px';
        el.style.top  = (y - ch / 2) + 'px';
        el.style.transformOrigin = 'center center';
        el.style.transform = `scale(${k})`;

        this._coinX = x;
        this._coinY = y;

        if (p >= 1) return resolve();
        requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
    }).then(() => {
		this._flying = false;
	});
  }

  async _fadeStripOut(duration) {
    return new Promise(resolve => {
      if (!this._stripEl) return resolve();
      const el = this._stripEl;
      const t0 = performance.now();
      const tick = () => {
        if (this._cancelled) return resolve();
        const p = Math.min(1, (performance.now() - t0) / duration);
        el.style.opacity = String(1 - p);
        if (p >= 1) {
          el.remove();
          this._stripEl = null;
          this._labelEl = null;
          return resolve();
        }
        requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
    });
  }

  /* ============================================================
   *  RESIZE
   * ============================================================ */

  _measureCoinSize() {
    const scale = uiScale.scale || 1;
    this._coinW = Math.round(COIN_W_REF * scale);
    this._coinH = Math.round(COIN_H_REF * scale);
  }

  _onSceneResize() {
    // не дёргаем во время полёта — траектория уже строится от старых координат
    if (this._flying) return;

    const el = this._coinEl || this._tokenEl;
    if (!el) return;

    this._measureCoinSize();
    el.style.width  = this._coinW + 'px';
    el.style.height = this._coinH + 'px';

    if (this._cornerSide) {
      this._positionInCorner(this._cornerSide);
    } else if (this._layerEl) {
      // в центре — просто центрируем заново
      const W = this.scene.width, H = this.scene.height;
      this._coinX = W / 2;
      this._coinY = H / 2;
      el.style.left = (this._coinX - this._coinW / 2) + 'px';
      el.style.top  = (this._coinY - this._coinH / 2) + 'px';
      el.style.transform = '';
    }
  }

  _positionInCorner(side) {
    const el = this._coinEl || this._tokenEl;
    if (!el) return;

    const W = this.scene.width, H = this.scene.height;
    const C = SceneLayout.config;
    const frameT = C.frameT * H;
    const frameR = W - C.frameR * W;
    const frameB = H - C.frameB * H;

    const kEnd  = 0.8;
    const cwEnd = this._coinW * kEnd;
    const chEnd = this._coinH * kEnd;

    let endX, endY;
    if (side === 'cpu') {
      endX = frameR - cwEnd / 2;
      endY = frameT + chEnd / 2;
    } else {
      endX = frameR - cwEnd / 2;
      endY = frameB - chEnd / 2;
    }

    el.style.left = (endX - this._coinW / 2) + 'px';
    el.style.top  = (endY - this._coinH / 2) + 'px';
    el.style.transformOrigin = 'center center';
    el.style.transform = `scale(${kEnd})`;

    this._coinX = endX;
    this._coinY = endY;
  }

  _sleep(ms) {
    return new Promise(r => setTimeout(r, ms / this._speedMult));
  }
}
