const SUIT_GLYPH = {
  diamonds: '♦',
  hearts:   '♥',
  clubs:    '♣',
  spades:   '♠',
};

export class HUD {
  constructor() {
    this.el = document.createElement('div');
    this.el.id = 'hud';
    this.el.style.cssText = `
      position: fixed; bottom: 12px; left: 12px; z-index: 150;
      padding: 10px 14px; min-width: 210px;
      background: rgba(20,20,20,0.82);
      color: #eee; font: 13px/1.45 ui-monospace, Menlo, Consolas, monospace;
      border: 1px solid #444; border-radius: 6px;
      box-shadow: 0 4px 14px rgba(0,0,0,0.4);
      pointer-events: none; user-select: none;
    `;
    document.body.appendChild(this.el);
    this.set({});
  }

  set({ round, firstMove, attack, move, deckLeft, cpuHand }) {
    const fmt = (v) => v === 'player' ? 'Player' : v === 'cpu' ? 'CPU' : '—';

    const handStr = (cpuHand && cpuHand.length)
      ? cpuHand.map(c => `${c.rank}${SUIT_GLYPH[c.suit] || '?'}`).join(',') + ','
      : '—';

    this.el.innerHTML = `
      <div style="display:flex; justify-content:space-between; gap:12px;">
        <span style="opacity:.7;">Round</span><b>${round ?? '—'}</b>
      </div>
      <div style="display:flex; justify-content:space-between; gap:12px;">
        <span style="opacity:.7;">Deck</span><b>${deckLeft ?? '—'}</b>
      </div>
      <div style="height:1px; background:#444; margin:6px 0;"></div>
      <div style="display:flex; justify-content:space-between; gap:12px;">
        <span>👑 First</span><b>${fmt(firstMove)}</b>
      </div>
      <div style="display:flex; justify-content:space-between; gap:12px;">
        <span>⚔ Attack</span><b>${fmt(attack)}</b>
      </div>
      <div style="display:flex; justify-content:space-between; gap:12px;">
        <span>→ Move</span><b>${fmt(move)}</b>
      </div>
      <div style="height:1px; background:#444; margin:6px 0;"></div>
      <div style="font-size:12px; color:#bbb; word-break:break-all;">
        <b style="opacity:.7;">CPU:</b> ${handStr}
      </div>
    `;
  }

  setHidden(v) {
    this.el.style.display = v ? 'none' : 'block';
  }
  
  setDebugOnly(v) {
    this._debugOnly = !!v;
    if (this._debugOnly) this.setHidden(true);
  }
}
