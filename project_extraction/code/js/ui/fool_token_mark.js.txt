import { ASSETS } from '../assets.js';
import { Progress } from '../game/progress.js';

/** Есть ли жетон у указанной стороны. */
export function hasToken(side) {
  const t = Progress.foolToken;
  return !!(t && t.owner === side);
}

/**
 * Повесить жетон на элемент. corner: 'tl' | 'tr'.
 * Если у указанной стороны нет жетона — ничего не создаёт,
 * но старые метки в этом элементе убирает.
 *
 * @param {HTMLElement} hostEl
 * @param {Object} opts
 * @param {'player'|'cpu'} opts.side
 * @param {'tl'|'tr'} [opts.corner='tl']
 * @param {string} [opts.heightCss] — CSS-выражение высоты (по умолчанию calc(77px * var(--ui)))
 * @param {string} [opts.mode] — если задан, метка ставится только для player.token.mode === mode
 */
export function attachTokenMark(hostEl, opts = {}) {
  if (!hostEl) return null;

  // убираем старые метки в этом хосте
  hostEl.querySelectorAll(':scope > .fool-token').forEach(el => el.remove());

  const { side, corner = 'tl', heightCss, mode } = opts;
  const t = Progress.foolToken;
  if (!t || t.owner !== side) return null;
  if (mode !== undefined && t.mode !== mode) return null;

  const img = document.createElement('img');
  img.className = `fool-token fool-token--${corner}`;
  img.src = ASSETS.foolToken;
  img.alt = '';
  img.draggable = false;
  if (heightCss) {
    img.style.height = heightCss;
    img.style.width = 'auto';
  }
  hostEl.appendChild(img);
  return img;
}

/** Убрать все метки в поддереве. */
export function clearTokenMarks(root = document) {
  root.querySelectorAll('.fool-token').forEach(el => el.remove());
}
