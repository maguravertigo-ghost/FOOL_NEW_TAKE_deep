  const defaults = m.DEFAULTS;

  console.log('=== “≈ ”ў»… SceneLayout.config ===');

  });
  console.log('export const DEFAULTS = {\n' + lines.join('\n') + '\n};');

  console.log('\n=== «начени€ из localStorage ===');

})();
